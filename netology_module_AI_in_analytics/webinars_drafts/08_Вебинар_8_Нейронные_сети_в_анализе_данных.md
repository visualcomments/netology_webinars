### Полносвязная сеть для кредитного скоринга

```python
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import roc_auc_score, accuracy_score, classification_report

class NeuralNetwork:
    def __init__(self, layer_sizes, learning_rate=0.01, l2_reg=0.001):
        self.layer_sizes = layer_sizes
        self.lr = learning_rate
        self.l2_reg = l2_reg
        self.weights = []
        self.biases = []

        for i in range(len(layer_sizes) - 1):
            w = np.random.randn(layer_sizes[i], layer_sizes[i+1]) * np.sqrt(2.0 / layer_sizes[i])
            b = np.zeros((1, layer_sizes[i+1]))
            self.weights.append(w)
            self.biases.append(b)

    def relu(self, x):
        return np.maximum(0, x)

    def relu_derivative(self, x):
        return (x > 0).astype(float)

    def sigmoid(self, x):
        x = np.clip(x, -500, 500)
        return 1 / (1 + np.exp(-x))

    def forward(self, X):
        self.activations = [X]
        self.z_values = []
        current = X
        for i in range(len(self.weights) - 1):
            z = current @ self.weights[i] + self.biases[i]
            self.z_values.append(z)
            current = self.relu(z)
            self.activations.append(current)
        z = current @ self.weights[-1] + self.biases[-1]
        self.z_values.append(z)
        output = self.sigmoid(z)
        self.activations.append(output)
        return output

    def backward(self, X, y):
        m = X.shape[0]
        d_weights = [None] * len(self.weights)
        d_biases = [None] * len(self.biases)
        output = self.activations[-1]
        delta = output - y.reshape(-1, 1)
        for i in range(len(self.weights) - 1, -1, -1):
            d_weights[i] = (self.activations[i].T @ delta) / m + self.l2_reg * self.weights[i]
            d_biases[i] = np.sum(delta, axis=0, keepdims=True) / m
            if i > 0:
                delta = (delta @ self.weights[i].T) * self.relu_derivative(self.z_values[i-1])
        for i in range(len(self.weights)):
            self.weights[i] -= self.lr * d_weights[i]
            self.biases[i] -= self.lr * d_biases[i]

    def train(self, X, y, epochs=100, batch_size=32, verbose=True):
        losses = []
        for epoch in range(epochs):
            indices = np.random.permutation(X.shape[0])
            X_shuffled = X[indices]
            y_shuffled = y[indices]
            epoch_loss = 0
            n_batches = 0
            for start in range(0, X.shape[0], batch_size):
                end = min(start + batch_size, X.shape[0])
                X_batch = X_shuffled[start:end]
                y_batch = y_shuffled[start:end]
                output = self.forward(X_batch)
                eps = 1e-8
                loss = -np.mean(y_batch.reshape(-1, 1) * np.log(output + eps) +
                               (1 - y_batch.reshape(-1, 1)) * np.log(1 - output + eps))
                epoch_loss += loss
                n_batches += 1
                self.backward(X_batch, y_batch)
            avg_loss = epoch_loss / n_batches
            losses.append(avg_loss)
            if verbose and (epoch + 1) % 20 == 0:
                y_pred_proba = self.forward(X).flatten()
                auc = roc_auc_score(y, y_pred_proba)
                print(f"  Epoch {epoch+1}/{epochs} | Loss: {avg_loss:.4f} | AUC: {auc:.4f}")
        return losses

    def predict_proba(self, X):
        return self.forward(X).flatten()

    def predict(self, X, threshold=0.5):
        return (self.predict_proba(X) >= threshold).astype(int)

# === Данные для кредитного скоринга ===
np.random.seed(42)
n = 10000

data = {
    'age': np.random.randint(18, 70, n),
    'income': np.random.lognormal(10, 0.8, n),
    'loan_amount': np.random.lognormal(9, 1, n),
    'credit_score': np.random.randint(300, 850, n),
    'employment_years': np.random.randint(0, 30, n),
    'num_open_accounts': np.random.randint(0, 15, n),
    'delinquencies_2y': np.random.poisson(0.3, n),
    'credit_utilization': np.random.uniform(0, 1, n),
    'debt_to_income': np.random.uniform(0, 0.8, n),
    'num_inquiries_6m': np.random.poisson(1, n),
}

df = pd.DataFrame(data)

default_prob = (
    0.5
    - 0.0001 * df['income']
    + 0.00001 * df['loan_amount']
    - 0.001 * df['credit_score']
    + 0.15 * df['delinquencies_2y']
    + 0.4 * df['credit_utilization']
    + 0.3 * df['debt_to_income']
    + np.random.normal(0, 0.12, n)
)
df['default'] = (default_prob > 0.35).astype(int)

X = df.drop('default', axis=1).values

print(f"Train: {X_train.shape}, Test: {X_test.shape}")
print(f"Default rate: {y.mean():.2%}")
```

## 2. LSTM для прогнозирования временных рядов

```python
class SimpleLSTM:
    def __init__(self, input_size, hidden_size):
        self.hidden_size = hidden_size
        self.Wf = np.random.randn(input_size + hidden_size, hidden_size) * 0.1
        self.Wi = np.random.randn(input_size + hidden_size, hidden_size) * 0.1
        self.Wc = np.random.randn(input_size + hidden_size, hidden_size) * 0.1
        self.Wo = np.random.randn(input_size + hidden_size, hidden_size) * 0.1
        self.bf = np.zeros((1, hidden_size))
        self.bi = np.zeros((1, hidden_size))
        self.bc = np.zeros((1, hidden_size))
        self.bo = np.zeros((1, hidden_size))

    def sigmoid(self, x):
        return 1 / (1 + np.exp(-np.clip(x, -500, 500)))

    def forward_step(self, x_t, h_prev, c_prev):
        combined = np.hstack([h_prev, x_t.reshape(1, -1)])
        f = self.sigmoid(combined @ self.Wf + self.bf)
        i = self.sigmoid(combined @ self.Wi + self.bi)
        c_candidate = np.tanh(combined @ self.Wc + self.bc)
        o = self.sigmoid(combined @ self.Wo + self.bo)
        c_new = f * c_prev + i * c_candidate
        h_new = o * np.tanh(c_new)
        return h_new, c_new


class LSTMForecaster:
    def __init__(self, input_size, hidden_size=32, output_size=1, learning_rate=0.001):
        self.lstm = SimpleLSTM(input_size, hidden_size)
        self.hidden_size = hidden_size
        self.lr = learning_rate
        self.W_out = np.random.randn(hidden_size, output_size) * 0.1
        self.b_out = np.zeros((1, output_size))

    def forward_sequence(self, sequence):
        h = np.zeros((1, self.hidden_size))
        c = np.zeros((1, self.hidden_size))
        for x_t in sequence:
            h, c = self.lstm.forward_step(x_t, h, c)
        prediction = h @ self.W_out + self.b_out
        return prediction.flatten(), h, c

    def train_on_sequence(self, sequences, targets, epochs=50):
        for epoch in range(epochs):
            total_loss = 0
            for seq, target in zip(sequences, targets):
                prediction, h, c = self.forward_sequence(seq)
                error = prediction - target
                loss = np.mean(error ** 2)
                total_loss += loss
                gradient = error.reshape(-1, 1) @ h
                self.W_out -= self.lr * gradient
                self.b_out -= self.lr * error.reshape(1, -1)
            if (epoch + 1) % 10 == 0:
                print(f"  Epoch {epoch+1}/{epochs} | Loss: {total_loss/len(sequences):.6f}")

# === Данные временного ряда ===

n_days = 365

trend = 0.05 * t
seasonality = 10 * np.sin(2 * np.pi * t / 30)

for i in range(len(prices) - window_size - 1):

    seq_norm = (seq - seq.mean()) / (seq.std() + 1e-8)
    target = prices[i + window_size]
    sequences.append(seq_norm)

split = int(len(sequences) * 0.8)

predictions = []
```

## 3. CNN для классификации документов

```python
class SimpleCNN:
    def __init__(self, input_shape, n_filters=8, filter_size=3, n_classes=4):
        self.n_filters = n_filters
        self.filter_size = filter_size
        self.n_classes = n_classes
        self.filters = np.random.randn(filter_size, filter_size, n_filters) * 0.1
        self.bias_conv = np.zeros(n_filters)
        conv_h = input_shape[0] - filter_size + 1
        conv_w = input_shape[1] - filter_size + 1
        pool_h = conv_h // 2
        pool_w = conv_w // 2
        fc_input = pool_h * pool_w * n_filters
        self.W_fc = np.random.randn(fc_input, n_classes) * 0.1
        self.b_fc = np.zeros(n_classes)
        self.lr = 0.01

    def conv2d(self, image):
        h, w = image.shape
        conv_h = h - self.filter_size + 1
        conv_w = w - self.filter_size + 1
        output = np.zeros((conv_h, conv_w, self.n_filters))
        for f in range(self.n_filters):
            for i in range(conv_h):
                for j in range(conv_w):
                    patch = image[i:i+self.filter_size, j:j+self.filter_size]
                    output[i, j, f] = np.sum(patch * self.filters[:, :, f]) + self.bias_conv[f]
        return output

    def max_pool2d(self, x, pool_size=2):
        h, w, c = x.shape
        pool_h = h // pool_size
        pool_w = w // pool_size
        output = np.zeros((pool_h, pool_w, c))
        for i in range(pool_h):
            for j in range(pool_w):
                for f in range(c):
                    patch = x[i*pool_size:(i+1)*pool_size, j*pool_size:(j+1)*pool_size, f]
                    output[i, j, f] = patch.max()
        return output

    def relu(self, x):
        return np.maximum(0, x)

    def softmax(self, x):
        exp_x = np.exp(x - np.max(x))
        return exp_x / exp_x.sum()

    def forward(self, image):
        conv_out = self.conv2d(image)
        relu_out = self.relu(conv_out)
        pool_out = self.max_pool2d(relu_out)
        flattened = pool_out.flatten()
        logits = flattened @ self.W_fc + self.b_fc
        probs = self.softmax(logits)
        return probs, flattened

    def predict(self, image):
        probs, _ = self.forward(image)
        return np.argmax(probs), probs
```

## 4. Autoencoder для обнаружения аномалий

```python
class Autoencoder:
    def __init__(self, input_size, encoding_size=8, learning_rate=0.01):
        self.lr = learning_rate
        self.W_enc = np.random.randn(input_size, encoding_size) * np.sqrt(2.0 / input_size)
        self.b_enc = np.zeros(encoding_size)
        self.W_dec = np.random.randn(encoding_size, input_size) * np.sqrt(2.0 / encoding_size)
        self.b_dec = np.zeros(input_size)

    def relu(self, x):
        return np.maximum(0, x)

    def encode(self, X):
        return self.relu(X @ self.W_enc + self.b_enc)

    def decode(self, Z):
        return Z @ self.W_dec + self.b_dec

    def forward(self, X):
        Z = self.encode(X)
        X_reconstructed = self.decode(Z)
        return X_reconstructed, Z

    def train(self, X, epochs=100, batch_size=32):
        losses = []
        for epoch in range(epochs):
            indices = np.random.permutation(X.shape[0])
            X_shuffled = X[indices]
            epoch_loss = 0
            n_batches = 0
            for start in range(0, X.shape[0], batch_size):
                end = min(start + batch_size, X.shape[0])
                X_batch = X_shuffled[start:end]
                X_recon, Z = self.forward(X_batch)
                error = X_recon - X_batch
                loss = np.mean(error ** 2)
                epoch_loss += loss
                n_batches += 1
                dW_dec = Z.T @ error / X_batch.shape[0]
                db_dec = np.mean(error, axis=0)
                dZ = error @ self.W_dec.T
                dZ = dZ * (Z > 0).astype(float)
                dW_enc = X_batch.T @ dZ / X_batch.shape[0]
                db_enc = np.mean(dZ, axis=0)
                self.W_dec -= self.lr * dW_dec
                self.b_dec -= self.lr * db_dec
                self.W_enc -= self.lr * dW_enc
                self.b_enc -= self.lr * db_enc
            avg_loss = epoch_loss / n_batches
            losses.append(avg_loss)
            if (epoch + 1) % 20 == 0:
                print(f"  Epoch {epoch+1}/{epochs} | Loss: {avg_loss:.6f}")
        return losses

    def reconstruction_error(self, X):
        X_recon, _ = self.forward(X)
        return np.mean((X - X_recon) ** 2, axis=1)
```

```python
# === Полный пайплайн: от данных до прогноза ===

class FintechMLPipeline:
    """Универсальный пайплайн для финтех задач"""

    def __init__(self, task_type='classification'):
        self.task_type = task_type
        self.scaler = StandardScaler()
        self.model = None

    def prepare_data(self, df, target_col, test_size=0.25):
        X = df.drop(target_col, axis=1)
        y = df[target_col]

        # Автоматическое кодирование категориальных признаков
        cat_cols = X.select_dtypes(include=['object', 'category']).columns
        if len(cat_cols) > 0:
            X = pd.get_dummies(X, columns=cat_cols, drop_first=True)

        X_train, X_test, y_train, y_test = train_test_split(
            X, y, test_size=test_size, random_state=42, stratify=y
        )

        X_train_scaled = self.scaler.fit_transform(X_train)
        X_test_scaled = self.scaler.transform(X_test)

        return X_train_scaled, X_test_scaled, y_train.values, y_test.values, X.columns

    def train_classical(self, X_train, y_train):
        self.model = GradientBoostingClassifier(
            n_estimators=200, max_depth=5, learning_rate=0.1, random_state=42
        )
        self.model.fit(X_train, y_train)
        return self.model

    def train_neural(self, X_train, y_train, layer_sizes=None):
        if layer_sizes is None:
            layer_sizes = [X_train.shape[1], 64, 32, 1]
        self.model = NeuralNetwork(layer_sizes, learning_rate=0.01)
        self.model.train(X_train, y_train, epochs=100, batch_size=64, verbose=False)
        return self.model

    def evaluate(self, X_test, y_test):
        if isinstance(self.model, NeuralNetwork):
            y_proba = self.model.predict_proba(X_test)
        else:
            y_proba = self.model.predict_proba(X_test)[:, 1]

        auc = roc_auc_score(y_test, y_proba)
        y_pred = (y_proba >= 0.5).astype(int)
        acc = accuracy_score(y_test, y_pred)

        return {'ROC-AUC': auc, 'Accuracy': acc, 'predictions': y_proba}

    def feature_importance(self, feature_names):
        if isinstance(self.model, NeuralNetwork):
            # Approximate через перестановку
            return None
        else:
            importance = pd.DataFrame({
                'feature': feature_names,
                'importance': self.model.feature_importances_
            }).sort_values('importance', ascending=False)
            return importance


# === Использование пайплайна ===
```

## Практические рекомендации

1. **Нейросеть vs классика:**
   - Табличные данные: Gradient Boosting часто лучше и быстрее
   - НС выигрывает на изображениях, тексте, последовательностях
   - Для малых данных — классические методы надёжнее

2. **Архитектура НС:**
   - Начинайте с простой архитектуры, усложняйте постепенно
   - Batch normalization ускоряет сходимость
   - Dropout предотвращает переобучение
   - Early stopping по валидационной метрике

3. **Обучение:**
   - Learning rate scheduling (cosine, step decay)
   - Adam optimizer обычно лучше SGD
   - Градиентный клиппинг для предотвращения взрывов
   - Нормализация входных данных обязательна

4. **Production:**
   - ONNX для деплоя моделей
   - Квантизация для ускорения инференса
   - Мониторинг дрейфа данных
   - A/B тестирование новых моделей
