### 2. Sentiment Analysis (анализ тональности)

```python
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import classification_report, accuracy_score
from sklearn.svm import SVC

# === Sentiment: положительный (4-5) vs отрицательный (1-2) ===
reviews_df['sentiment'] = reviews_df['rating'].apply(lambda x: 1 if x >= 4 else 0)

print(f"Положительных: {(reviews_df['sentiment'] == 1).sum()}")
print(f"Отрицательных: {(reviews_df['sentiment'] == 0).sum()}")

# === TF-IDF векторизация ===
tfidf = TfidfVectorizer(
    max_features=1000,
    ngram_range=(1, 2),  # Униграммы + биграммы
    min_df=1,
    max_df=0.95,
    sublinear_tf=True
)

X_tfidf = tfidf.fit_transform(reviews_df['clean_text'])
y_sentiment = reviews_df['sentiment']

X_train, X_test, y_train, y_test = train_test_split(
    X_tfidf, y_sentiment, test_size=0.3, random_state=42, stratify=y_sentiment
)

# === Логистическая регрессия ===
lr = LogisticRegression(random_state=42, C=1.0)
lr.fit(X_train, y_train)

y_pred = lr.predict(X_test)
print(f"\nLogistic Regression Accuracy: {accuracy_score(y_test, y_pred):.4f}")
print(classification_report(y_test, y_pred, target_names=['Negative', 'Positive']))

# === SVM ===
svm = SVC(kernel='linear', random_state=42)
svm.fit(X_train, y_train)
y_pred_svm = svm.predict(X_test)
print(f"\nSVM Accuracy: {accuracy_score(y_test, y_pred_svm):.4f}")
print(classification_report(y_test, y_pred_svm, target_names=['Negative', 'Positive']))

# === Важность признаков (слов) ===
feature_names = tfidf.get_feature_names_out()
coefficients = lr.coef_[0]

top_positive = sorted(zip(feature_names, coefficients), key=lambda x: -x[1])[:10]
top_negative = sorted(zip(feature_names, coefficients), key=lambda x: x[1])[:10]

print("\nСлова, ассоциированные с положительным отзывом:")
for word, coef in top_positive:
    print(f"  {word}: {coef:.3f}")

print("\nСлова, ассоциированные с отрицательным отзывом:")
for word, coef in top_negative:
    print(f"  {word}: {coef:.3f}")
```

### 3. Тематическое моделирование (Topic Modeling)

```python
# === LDA: автоматическое выявление тем в отзывах ===
from sklearn.decomposition import LatentDirichletAllocation

# Используем CountVectorizer для LDA
from sklearn.feature_extraction.text import CountVectorizer

count_vec = CountVectorizer(
    max_features=500,
    ngram_range=(1, 1),
    min_df=1,
    stop_words=list(stop_words)
)

X_counts = count_vec.fit_transform(reviews_df['clean_text'])

# LDA с 4 темами
n_topics = 4
lda = LatentDirichletAllocation(n_components=n_topics, random_state=42, max_iter=20)
lda.fit(X_counts)

feature_names = count_vec.get_feature_names_out()

print("=" * 60)
print("ТЕМЫ В ОТЗЫВАХ КЛИЕНТОВ")
print("=" * 60)

for topic_idx, topic in enumerate(lda.components_):
    top_words = [feature_names[i] for i in topic.argsort()[:-10 - 1:-1]]
    print(f"\nТема {topic_idx + 1}: {', '.join(top_words)}")

# === Распределение тем по отзывам ===
topic_distribution = lda.transform(X_counts)
reviews_df['dominant_topic'] = topic_distribution.argmax(axis=1)

print("\nРаспределение тем:")
topic_counts = reviews_df['dominant_topic'].value_counts().sort_index()
for topic, count in topic_counts.items():
    print(f"  Тема {topic + 1}: {count} отзывов ({count/len(reviews_df)*100:.0f}%)")

# === Связь тем с рейтингом ===
print("\nСредний рейтинг по темам:")
for topic in range(n_topics):
    mask = reviews_df['dominant_topic'] == topic
    avg_rating = reviews_df.loc[mask, 'rating'].mean()
    print(f"  Тема {topic + 1}: {avg_rating:.1f}")
```

```python
np.random.seed(42)

news_data = {
    'headline': [

        "Безработица снизилась до минимума за 5 лет",

    ],
    'date': pd.date_range('2024-01-01', periods=10, freq='D'),
}

news_df = pd.DataFrame(news_data)

# === Ключевые слова и категории ===
finance_keywords = {
    'monetary': ['ставка', 'центральный банк', 'ключевая', 'денежный'],
    'positive': ['прибыль', 'рост', 'максимум', 'поддержка', 'снизилась'],
    'negative': ['дефолт', 'санкции', 'обвалилась', 'инфляция'],
    'market': ['рынок', 'акций', 'нефть', 'криптовалюта'],
}

def categorize_news(headline):
    headline_lower = headline.lower()
    categories = []
    for category, keywords in finance_keywords.items():
        if any(kw in headline_lower for kw in keywords):
            categories.append(category)
    return categories if categories else ['neutral']

news_df['categories'] = news_df['headline'].apply(categorize_news)

print("Категоризация новостей:")
for _, row in news_df.iterrows():
    print(f"  [{', '.join(row['categories'])}] {row['headline']}")

# === Sentiment score для новостей ===
positive_words = {'прибыль', 'рост', 'максимум', 'поддержка', 'снизилась', 'подорожала'}
negative_words = {'дефолт', 'санкции', 'обвалилась', 'инфляция', 'выросла', 'повысил'}

def news_sentiment(headline):
    words = set(headline.lower().split())
    pos = len(words & positive_words)
    neg = len(words & negative_words)
    if pos > neg:
        return 1
    elif neg > pos:
        return -1
    return 0

news_df['sentiment'] = news_df['headline'].apply(news_sentiment)

print("\nSentiment новостей:")
for _, row in news_df.iterrows():
    sentiment_label = {1: 'POS', -1: 'NEG', 0: 'NEU'}[row['sentiment']]
    print(f"  [{sentiment_label}] {row['headline']}")
```

### 5. Collaborative Filtering

```python
# === Рекомендация финансовых продуктов ===
# Matrix Factorization через SVD

from sklearn.decomposition import TruncatedSVD
from sklearn.metrics.pairwise import cosine_similarity

# Матрица клиент-продукт (1 = использует, 0 = нет, NaN = неизвестно)
np.random.seed(42)
n_clients = 20
n_products = 8

product_names = ['checking', 'savings', 'credit_card', 'mortgage',
                 'investment', 'insurance', 'loan', 'deposit']

# Генерация матрицы с паттернами
client_product_matrix = np.zeros((n_clients, n_products))

for i in range(n_clients):
    # Базовые продукты
    client_product_matrix[i, 0] = 1  # checking — у всех
    if np.random.random() < 0.7:
        client_product_matrix[i, 1] = 1  # savings
    if np.random.random() < 0.5:
        client_product_matrix[i, 2] = 1  # credit_card

    # Коррелированные продукты
    if client_product_matrix[i, 3] == 0 and np.random.random() < 0.3:
        client_product_matrix[i, 3] = 1  # mortgage
        if np.random.random() < 0.8:
            client_product_matrix[i, 5] = 1  # insurance (часто с mortgage)

    if np.random.random() < 0.25:
        client_product_matrix[i, 4] = 1  # investment
        if np.random.random() < 0.6:
            client_product_matrix[i, 7] = 1  # deposit

    if np.random.random() < 0.2:
        client_product_matrix[i, 6] = 1  # loan

# Добавляем пропуски (не все данные известны)
mask = np.random.random((n_clients, n_products)) < 0.15
client_product_matrix[mask] = np.nan

print("Матрица клиент-продукт (1=использует, 0=нет, NaN=неизвестно):")
print("       " + "  ".join(f"{p[:6]:>6}" for p in product_names))
for i in range(n_clients):
    row_str = f"Client{i+1:02d}: "
    for j in range(n_products):
        val = client_product_matrix[i, j]
        if np.isnan(val):
            row_str += "   ?  "
        else:
            row_str += f"  {int(val)}   "
    print(row_str)

# === SVD для Collaborative Filtering ===
# Заполняем NaN нулями для SVD
matrix_filled = np.nan_to_num(client_product_matrix, nan=0)

# SVD разложение
n_latent = 3
svd = TruncatedSVD(n_components=n_latent, random_state=42)
client_factors = svd.fit_transform(matrix_filled)
product_factors = svd.components_.T

print(f"\nSVD: объяснённая дисперсия = {svd.explained_variance_ratio_.sum():.2%}")

# === Предсказание: клиент -> продукт ===
# Score = client_factors @ product_factors.T
predicted_scores = client_factors @ product_factors.T

# Для каждого клиента — топ продуктов, которые он ещё не использует
print("\nРекомендации для клиентов:")
for i in range(min(5, n_clients)):
    used = set(np.where(client_product_matrix[i] == 1)[0])
    unknown = set(np.where(np.isnan(client_product_matrix[i]))[0])
    candidates = unknown - used

    if candidates:
        scores = [(j, predicted_scores[i, j]) for j in candidates]
        scores.sort(key=lambda x: -x[1])
        top_recs = scores[:2]
        print(f"  Client {i+1:02d}: рекомендую {[product_names[j] for j, _ in top_recs]}")
```

### 6. Content-Based рекомендации

```python
# === Рекомендации на основе характеристик клиента ===
# Если клиент похож на других — рекомендуем то, что используют похожие

from sklearn.neighbors import NearestNeighbors

# Характеристики клиентов
np.random.seed(42)
client_features = pd.DataFrame({
    'age': np.random.randint(18, 70, n_clients),
    'income': np.random.lognormal(10, 0.7, n_clients),
    'num_existing_products': client_product_matrix.sum(axis=1),
    'months_as_customer': np.random.randint(1, 60, n_clients),
    'digital_activity': np.random.uniform(0, 1, n_clients),
})

# Нормализация
from sklearn.preprocessing import StandardScaler
scaler = StandardScaler()
client_features_scaled = scaler.fit_transform(client_features)

# === KNN: поиск похожих клиентов ===
knn = NearestNeighbors(n_neighbors=5, metric='cosine')
knn.fit(client_features_scaled)

print("\nContent-Based рекомендации (похожие клиенты):")
for i in range(min(5, n_clients)):
    distances, indices = knn.kneighbors([client_features_scaled[i]], n_neighbors=4)
    similar_clients = indices[0][1:]  # Исключаем самого клиента

    # Какие продукты используют похожие клиенты, но не текущий?
    used_by_client = set(np.where(client_product_matrix[i] == 1)[0])
    recommended = Counter()

    for sim_idx in similar_clients:
        for j in range(n_products):
            if j not in used_by_client and client_product_matrix[sim_idx, j] == 1:
                recommended[j] += 1

    if recommended:
        top_recs = recommended.most_common(2)
        rec_names = [product_names[j] for j, _ in top_recs]
        print(f"  Client {i+1:02d} (возраст={client_features['age'].iloc[i]}, доход={client_features['income'].iloc[i]:,.0f}): {rec_names}")
```

### 7. Hybrid рекомендации

```python
# === Гибридная система: CF + Content-Based ===
def hybrid_recommendations(client_idx, cf_scores, knn_model, client_features_scaled,
                           client_product_matrix, product_names, alpha=0.5):
    """
    alpha: вес CF (1-alpha: вес content-based)
    """
    n_products = len(product_names)
    used = set(np.where(client_product_matrix[client_idx] == 1)[0])
    unknown = set(np.where(np.isnan(client_product_matrix[client_idx]))[0])
    candidates = unknown - used

    if not candidates:
        return []

    # CF score
    cf_score_dict = {j: cf_scores[client_idx, j] for j in candidates}

    # Content-based score (через похожих клиентов)
    distances, indices = knn_model.kneighbors([client_features_scaled[client_idx]], n_neighbors=4)
    similar_clients = indices[0][1:]

    cb_scores = {}
    for j in candidates:
        score = 0
        for sim_idx in similar_clients:
            if client_product_matrix[sim_idx, j] == 1:
                score += 1
        cb_scores[j] = score / len(similar_clients) if similar_clients else 0

    # Нормализация CF scores
    cf_vals = list(cf_score_dict.values())
    cf_min, cf_max = min(cf_vals), max(cf_vals)
    cf_range = cf_max - cf_min if cf_max > cf_min else 1
    cf_normalized = {j: (cf_score_dict[j] - cf_min) / cf_range for j in candidates}

    # Hybrid score
    hybrid_scores = {}
    for j in candidates:
        hybrid_scores[j] = alpha * cf_normalized[j] + (1 - alpha) * cb_scores[j]

    sorted_recs = sorted(hybrid_scores.items(), key=lambda x: -x[1])
    return [(product_names[j], score) for j, score in sorted_recs[:3]]


print("\nHybrid рекомендации (alpha=0.5):")
for i in range(min(5, n_clients)):
    recs = hybrid_recommendations(
        i, predicted_scores, knn, client_features_scaled,
        client_product_matrix, product_names, alpha=0.5
    )
    if recs:
        rec_str = ", ".join([f"{name}({score:.2f})" for name, score in recs])
        print(f"  Client {i+1:02d}: {rec_str}")
```

### 8. Оценка рекомендательной системы

```python
# === Метрики качества рекомендаций ===

def precision_at_k(recommended, relevant, k=3):
    """Precision@K: доля релевантных в топ-K"""
    rec_k = recommended[:k]
    hits = len(set(rec_k) & set(relevant))
    return hits / k if k > 0 else 0

def recall_at_k(recommended, relevant, k=3):
    """Recall@K: доля найденных релевантных"""
    rec_k = recommended[:k]
    hits = len(set(rec_k) & set(relevant))
    return hits / len(relevant) if relevant else 0

def ndcg_at_k(recommended, relevant, k=3):
    """NDCG@K: нормализованный DCG"""
    rec_k = recommended[:k]
    dcg = sum(1.0 / np.log2(i + 2) for i, item in enumerate(rec_k) if item in relevant)
    # Ideal DCG
    ideal_hits = min(len(relevant), k)
    idcg = sum(1.0 / np.log2(i + 2) for i in range(ideal_hits))
    return dcg / idcg if idcg > 0 else 0

# === Симуляция: скрываем часть данных и проверяем рекомендации ===
np.random.seed(42)

# Создаём train/test split для каждого клиента
train_matrix = client_product_matrix.copy()
test_relevant = {}

for i in range(n_clients):
    used_products = list(np.where(train_matrix[i] == 1)[0])
    if len(used_products) >= 2:
        # Скрываем 1 продукт для теста
        hidden_idx = np.random.choice(used_products)
        test_relevant[i] = {hidden_idx}
        train_matrix[i, hidden_idx] = np.nan

# Генерация рекомендаций на train матрице
train_filled = np.nan_to_num(train_matrix, nan=0)
svd_train = TruncatedSVD(n_components=3, random_state=42)
client_factors_train = svd_train.fit_transform(train_filled)
product_factors_train = svd_train.components_.T
cf_scores_train = client_factors_train @ product_factors_train.T

# Оценка
precisions = []
recalls = []
ndcgs = []

for i in test_relevant:
    used = set(np.where(train_matrix[i] == 1)[0])
    unknown = set(np.where(np.isnan(train_matrix[i]))[0])
    candidates = list(unknown - used)

    if not candidates:
        continue

    scores = [(j, cf_scores_train[i, j]) for j in candidates]
    scores.sort(key=lambda x: -x[1])
    recommended = [j for j, _ in scores[:5]]

    relevant = test_relevant[i]
    precisions.append(precision_at_k(recommended, relevant, k=3))
    recalls.append(recall_at_k(recommended, relevant, k=3))
    ndcgs.append(ndcg_at_k(recommended, relevant, k=3))

print(f"Precision@3: {np.mean(precisions):.4f}")
print(f"Recall@3:    {np.mean(recalls):.4f}")
print(f"NDCG@3:      {np.mean(ndcgs):.4f}")
```

## Практические рекомендации

### NLP:
1. **Для sentiment analysis:** начинайте с TF-IDF + Logistic Regression, переходите к трансформерам при необходимости
2. **Для тематического моделирования:** LDA хорошо работает на русском при правильной предобработке
3. **Для production:** используйте готовые модели (ruBERT, DeepPavlov)

### Рекомендательные системы:
1. **Cold start:** content-based лучше для новых клиентов, CF — для существующих
2. **Hybrid approach:** комбинируйте CF и content-based для лучшего качества
3. **A/B тестирование:** обязательно тестируйте рекомендации на реальных пользователях
4. **Business metrics:** конверсия, cross-sell ratio, revenue per client
