# Вебинар 7: NLP — Большие языковые модели (LLM) в финтехе

## Обзор темы

Большие языковые модели (LLM) — GPT, BERT, LLaMA и их аналоги — революционизируют финтех: автоматизация поддержки, анализ документов, генерация отчётов, извлечение данных, детекция мошенничества по тексту. Разберём практическое применение.

## 1. Архитектура трансформеров (кратко)

### Ключевые компоненты
- **Self-Attention:** механизм взвешивания важности слов в контексте
- **Multi-Head Attention:** несколько параллельных attention-голов
- **Positional Encoding:** информация о позиции токена
- **Feed-Forward Network:** нелинейное преобразование
- **Layer Normalization:** стабилизация обучения

### Типы моделей
- **Encoder-only** (BERT): понимание текста, классификация, NER
- **Decoder-only** (GPT): генерация текста
- **Encoder-Decoder** (T5, BART): перевод, суммаризация

```python
# === Извлечение финансовых сущностей из текста ===
import re

def extract_financial_entities(text):
    """Извлечение ключевых финансовых сущностей"""
    entities = {}

    # Денежные суммы
    money_patterns = [
        r'(\d+(?:\.\d+)?)\s*(?:млрд|млн|тыс)\.?\s*(?:руб|долл|евро|usd|rub)',
        r'\$\s*(\d+(?:\.\d+)?)\s*(?:млрд|млн|тыс)?\.?',
        r'(\d+(?:\.\d+)?)\s*руб',
    ]
    amounts = []
    for pattern in money_patterns:
        amounts.extend(re.findall(pattern, text, re.IGNORECASE))
    if amounts:
        entities['amounts'] = amounts

    # Проценты
    pct_pattern = r'(\d+(?:\.\d+)?)\s*%'
    entities['percentages'] = re.findall(pct_pattern, text)

    # Компании (упрощённо — по ключевым словам)
    company_indicators = ['банк', 'компания', 'холдинг', 'группа', 'корпорация', 'Сбербанк', 'Газпром', 'Лукойл', 'Яндекс']
    words = text.split()
    companies = []
    for i, word in enumerate(words):
        for indicator in company_indicators:
            if indicator.lower() in word.lower():
                # Берём слово и следующее (название)
                company = word
                if i + 1 < len(words) and words[i + 1][0].isupper():
                    company += ' ' + words[i + 1]
                companies.append(company)
    if companies:
        entities['companies'] = list(set(companies))

    # Даты
    date_pattern = r'(\d{1,2}\.\d{1,2}\.\d{2,4}|\d{1,2}\s+(?:января|февраля|марта|апреля|мая|июня|июля|августа|сентября|октября|ноября|декабря))'
    entities['dates'] = re.findall(date_pattern, text, re.IGNORECASE)

    # Финансовые метрики
    metric_keywords = ['прибыль', 'выручка', 'убыток', 'дивиденд', 'ставка', 'рейтинг', 'индекс', 'капитализация']
    found_metrics = [kw for kw in metric_keywords if kw in text.lower()]
    if found_metrics:
        entities['metrics'] = found_metrics

    return entities


# === Тестирование NER ===
test_texts = [
```

## 4. Суммаризация финансовых документов

```python
# === Extractive Summarization: выделение ключевых предложений ===

def extractive_summarize(text, n_sentences=3):
    """Упрощённая экстрактивная суммаризация"""
    import re
    from collections import Counter

    # Разбиение на предложения
    sentences = re.split(r'[.!?]+', text)
    sentences = [s.strip() for s in sentences if len(s.strip()) > 20]

    if len(sentences) <= n_sentences:
        return text

    # Токенизация
    words_per_sentence = [s.lower().split() for s in sentences]
    all_words = [w for sent in words_per_sentence for w in sent]

    # TF слов
    word_freq = Counter(all_words)
    max_freq = max(word_freq.values()) if word_freq else 1
    word_freq_normalized = {w: f / max_freq for w, f in word_freq.items()}

    # Stop words
    stop_words = {'и', 'в', 'не', 'на', 'за', 'по', 'с', 'от', 'к', 'о', 'а', 'но',
                  'же', 'бы', 'ли', 'для', 'из', 'у', 'до', 'при', 'во', 'со',
                  'это', 'что', 'как', 'все', 'так', 'его', 'её', 'их', 'он', 'она',
                  'они', 'мы', 'вы', 'я', 'ты', 'мне', 'тебе', 'себя', 'свой'}

    # Score каждого предложения
    sentence_scores = []
    for i, words in enumerate(words_per_sentence):
        score = sum(word_freq_normalized.get(w, 0) for w in words if w not in stop_words)
        # Бонус за позицию (первые предложения важнее)
        position_bonus = 1.0 / (1 + 0.1 * i)
        # Бонус за наличие чисел (финансовые данные)
        number_bonus = 1.5 if any(re.search(r'\d', w) for w in words) else 1.0
        sentence_scores.append((i, score * position_bonus * number_bonus))

    # Топ-N предложений
    sentence_scores.sort(key=lambda x: -x[1])
    top_indices = sorted([idx for idx, _ in sentence_scores[:n_sentences]])

    summary = '. '.join(sentences[i] for i in top_indices) + '.'
    return summary


# === Длинный финансовый текст ===
long_report = """
Сбербанк опубликовал финансовые результаты за третий квартал 2024 года.
Чистая прибыль банка составила 350 миллиардов рублей, что на 15% выше аналогичного периода прошлого года.
Рост прибыли был обусловлен увеличением кредитного портфеля на 12% и снижением стоимости риска.
Розничный кредитный портфель вырос на 18% благодаря высокому спросу на ипотеку и потребительские кредиты.
Корпоративный портфель увеличился на 8%, основными заёмщиками остались компании нефтегазового сектора.
Комиссионный доход банка вырос на 22% за счёт развития экосистемы и цифровых сервисов.
Мобильное приложение Сбербанка достигло 45 миллионов активных пользователей.
Рентабельность капитала (ROE) составила 24%, что является одним из лучших показателей в отрасли.
Совет директоров рекомендовал выплатить дивиденды в размере 35 рублей на акцию.
```

```python
def generate_financial_report(client_data):
    """Генерация персонального финансового отчёта"""

    name = client_data['name']
    income = client_data['income']
    expenses = client_data['expenses']
    savings = client_data['savings']
    investments = client_data['investments']
    debts = client_data['debts']
    credit_score = client_data['credit_score']
    goals = client_data['goals']

    savings_rate = (income - expenses) / income * 100 if income > 0 else 0
    debt_to_income = debts / income * 100 if income > 0 else 0
    emergency_months = savings / expenses if expenses > 0 else 0

    report = f"""
=== ПЕРСОНАЛЬНЫЙ ФИНАНСОВЫЙ ОТЧЁТ ===
Клиент: {name}

Ежемесячные расходы: {expenses:,.0f} руб.

Накопления: {savings:,.0f} руб.
Инвестиции: {investments:,.0f} руб.
Общие активы: {savings + investments:,.0f} руб.

--- ОБЯЗАТЕЛЬСТВА ---

Оценка: {'Отличный' if credit_score >= 750 else 'Хороший' if credit_score >= 650 else 'Средний' if credit_score >= 550 else 'Низкий'}

--- РЕКОМЕНДАЦИИ ---
"""

    # Автоматические рекомендации
    recommendations = []

    if savings_rate < 10:
        recommendations.append(f"• Увеличьте норму сбережений. Текущая {savings_rate:.1f}% — ниже рекомендуемых 20%.")
    elif savings_rate >= 20:
        recommendations.append(f"• Отличная норма сбережений ({savings_rate:.1f}%). Рассмотрите увеличение инвестиционной части.")

        recommendations.append(f"• Сформирована подушка безопасности на {emergency_months:.1f} мес. Рекомендуется минимум 3-6 месяцев расходов.")

        recommendations.append(f"• Высокая долговая нагрузка ({debt_to_income:.1f}%). Рекомендуется приоритетное погашение долгов.")

    if credit_score < 650:
        recommendations.append(f"• Кредитный скор {credit_score} ниже среднего. Рекомендации: снизьте утилизацию кредитных карт, не допускайте просрочек.")

    report += "\n\n=== КОНЕЦ ОТЧЁТА ==="
```

```python
# === Извлечение данных из кредитных заявок ===

def extract_loan_application(text):
    """Парсинг кредитной заявки"""
    import re

    extracted = {}

    # ФИО
    name_pattern = r'(?:ФИО|имя|заёмщик)[:\s]*([А-ЯЁ][а-яё]+\s+[А-ЯЁ][а-яё]+\s+[А-ЯЁ][а-яё]+)'
    name_match = re.search(name_pattern, text, re.IGNORECASE)
    if name_match:
        extracted['full_name'] = name_match.group(1)

    # Сумма кредита
    amount_pattern = r'(?:сумма|запрашиваемая сумма|кредит)[:\s]*(\d+(?:\s*\d+)*)\s*(?:руб|rub|₽)?'
    amount_match = re.search(amount_pattern, text, re.IGNORECASE)
    if amount_match:
        extracted['loan_amount'] = int(amount_match.group(1).replace(' ', ''))

    # Срок
    term_pattern = r'(?:срок|период)[:\s]*(\d+)\s*(?:мес|месяц|лет|год)'
    term_match = re.search(term_pattern, text, re.IGNORECASE)
    if term_match:
        extracted['term_months'] = int(term_match.group(1))

    # Доход
    income_pattern = r'(?:доход|заработная плата|зарплата)[:\s]*(\d+(?:\s*\d+)*)\s*(?:руб|rub|₽)?'
    income_match = re.search(income_pattern, text, re.IGNORECASE)
    if income_match:
        extracted['monthly_income'] = int(income_match.group(1).replace(' ', ''))

    # Цель кредита
    purpose_keywords = ['ипотека', 'автокредит', 'потребительский', 'бизнес', 'образование', 'ремонт']
    for keyword in purpose_keywords:
        if keyword in text.lower():
            extracted['purpose'] = keyword
            break

    # Место работы
    employer_pattern = r'(?:работодатель|место работы|компания)[:\s]*([А-ЯЁ][а-яё]+(?:\s+[А-ЯЁ][а-яё]+)*)'
    employer_match = re.search(employer_pattern, text, re.IGNORECASE)
    if employer_match:
        extracted['employer'] = employer_match.group(1)

    return extracted


# === Тест ===
application_text = """
Заявка на кредит
ФИО: Иванов Алексей Сергеевич
Сумма кредита: 1 500 000 руб
Срок: 60 мес
Цель: ипотека
Ежемесячный доход: 180 000 руб
Работодатель: Газпром Нефть
Стаж работы: 7 лет
Кредитная история: без просрочек
"""

extracted = extract_loan_application(application_text)
print("Извлечённые данные из заявки:")
for key, value in extracted.items():
    print(f"  {key}: {value}")

# === Автоматическое предварительное решение ===
def preliminary_decision(data):
    if 'loan_amount' not in data or 'monthly_income' not in data:
        return "Недостаточно данных для решения"

    dti = data['loan_amount'] / (data['monthly_income'] * 12) * 100

    if dti > 50:
        decision = "Отказ: высокая долговая нагрузка"
    elif data.get('monthly_income', 0) < 30000:
        decision = "Отказ: недостаточный доход"
    elif dti < 30:
        decision = "Предварительно одобрено"
    else:
        decision = "Требуется дополнительная проверка"

    return f"{decision} (DTI: {dti:.1f}%)"


decision = preliminary_decision(extracted)
print(f"\nПредварительное решение: {decision}")
```

```python
                'response': "Доступные кредитные продукты: потребительский (от 5.9%), ипотека (от 4.5%), автокредит (от 3.9%). Какой вас интересует?"
            },
            'card': {
                'keywords': ['карта', 'блокиров', 'потерял карту'],
                'response': "Для блокировки карты нажмите 1. Для заказа новой — 2. Для изменения лимита — 3."

            'rate': {

            },
            'greeting': {
                'keywords': ['здравствуй', 'привет', 'добрый день', 'доброе утро'],
                'response': "Здравствуйте! Я ваш финансовый ассистент. Чем могу помочь?"
            },
        }

    def process_message(self, message):
        message_lower = message.lower()

        for intent, data in self.intents.items():
            if any(kw in message_lower for kw in data['keywords']):
                return data['response']

        return "Извините, я не понял запрос. Попробуйте переформулировать или выберите из меню: баланс, перевод, кредит, карта."
```

## Практические рекомендации

1. **Когда использовать LLM:**
   - Сложные NLP задачи (понимание контекста, генерация)
   - Работа с неструктурированными документами
   - Персонализация коммуникации

2. **Ограничения и риски:**
   - Галлюцинации (выдуманные факты) — критично в финансах
   - Стоимость API вызовов при большом объёме
   - Задержки (latency) для real-time задач
   - Безопасность данных (не отправлять PII в публичные API)

3. **Best Practices:**
   - RAG (Retrieval-Augmented Generation) для работы с актуальными данными
   - Валидация выводов LLM через правила
   - Fine-tuning на доменных данных для лучшего качества
   - Human-in-the-loop для критичных решений
