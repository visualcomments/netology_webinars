# Вебинар 4: Машинное обучение без учителя

## Обзор темы

Обучение без учителя (Unsupervised Learning) решает задачи, где нет размеченных данных. В финтехе это: сегментация клиентов, обнаружение аномалий и мошенничества, снижение размерности, выявление скрытых паттернов.

## 1. Кластеризация клиентов

### K-Means: сегментация по поведению

```python
import pandas as pd
import numpy as np
from sklearn.cluster import KMeans, DBSCAN, AgglomerativeClustering
from sklearn.preprocessing import StandardScaler, RobustScaler
from sklearn.metrics import silhouette_score, calinski_harabasz_score
from sklearn.decomposition import PCA
import matplotlib.pyplot as plt

# === Синтетические данные клиентов банка ===
np.random.seed(42)
n = 5000

clients = {
    'age': np.random.randint(18, 75, n),
    'income': np.random.lognormal(10, 0.8, n),
    'monthly_spending': np.random.lognormal(8, 0.7, n),
    'savings_balance': np.random.lognormal(7, 1.5, n),
    'num_transactions_month': np.random.poisson(30, n),
    'avg_transaction_amount': np.random.lognormal(4, 1, n),
    'num_products': np.random.randint(1, 8, n),
    'months_as_customer': np.random.randint(1, 120, n),
    'online_banking_usage': np.random.uniform(0, 1, n),
    'credit_card_debt': np.random.lognormal(6, 1.2, n),
}

df = pd.DataFrame(clients)

# === Предобработка ===
scaler = RobustScaler()  # Устойчив к выбросам
X_scaled = scaler.fit_transform(df)

# === Определение оптимального числа кластеров ===
inertias = []
silhouette_scores = []
K_range = range(2, 11)

for k in K_range:
    km = KMeans(n_clusters=k, random_state=42, n_init=10)
    labels = km.fit_predict(X_scaled)
    inertias.append(km.inertia_)
    silhouette_scores.append(silhouette_score(X_scaled, labels))

print("Метод локтя (inertia):")
for k, inertia in zip(K_range, inertias):
    print(f"  K={k}: inertia={inertia:,.0f}")

print("\nSilhouette Score:")
for k, score in zip(K_range, silhouette_scores):
    print(f"  K={k}: silhouette={score:.4f}")

# Оптимальное K (по silhouette)
optimal_k = K_range[np.argmax(silhouette_scores)]
print(f"\nОптимальное K: {optimal_k}")
```

## 2. Обнаружение аномалий (Anomaly Detection)

### Isolation Forest для Fraud Detection

```python
from sklearn.ensemble import IsolationForest
from sklearn.covariance import EllipticEnvelope
from sklearn.neighbors import LocalOutlierFactor
from sklearn.metrics import precision_score, recall_score, f1_score

# === Данные транзакций ===
np.random.seed(42)
n_normal = 9500
n_fraud = 500

# Нормальные транзакции
normal = {
    'amount': np.random.lognormal(3, 0.8, n_normal),
    'hour': np.random.choice(range(6, 23), n_normal),
    'distance_km': np.random.exponential(3, n_normal),
    'time_since_last': np.random.exponential(120, n_normal),
    'is_foreign': np.random.choice([0, 1], n_normal, p=[0.95, 0.05]),
    'is_online': np.random.choice([0, 1], n_normal, p=[0.7, 0.3]),
    'merchant_category': np.random.randint(1, 20, n_normal),
}

# Мошеннические транзакции (аномальные паттерны)
fraud = {
    'amount': np.random.lognormal(5, 1.2, n_fraud),  # Необычно крупные
    'hour': np.random.choice([0, 1, 2, 3, 4, 5], n_fraud),  # Ночью
    'distance_km': np.random.exponential(50, n_fraud),  # Далеко от дома
    'time_since_last': np.random.exponential(5, n_fraud),  # Быстро после предыдущей
    'is_foreign': np.random.choice([0, 1], n_fraud, p=[0.5, 0.5]),
    'is_online': np.random.choice([0, 1], n_fraud, p=[0.3, 0.7]),
    'merchant_category': np.random.randint(1, 20, n_fraud),
}

df_normal = pd.DataFrame(normal)
df_normal['is_fraud'] = 0

df_fraud = pd.DataFrame(fraud)
df_fraud['is_fraud'] = 1

transactions = pd.concat([df_normal, df_fraud], ignore_index=True)
transactions = transactions.sample(frac=1, random_state=42).reset_index(drop=True)

print(f"Доля мошенничества: {transactions['is_fraud'].mean():.2%}")

X_trans = transactions.drop('is_fraud', axis=1)
y_trans = transactions['is_fraud']

# === Isolation Forest ===
iso_forest = IsolationForest(
    n_estimators=200,
    contamination=0.05,  # Ожидаемая доля аномалий
    max_samples='auto',
    random_state=42,
    n_jobs=-1
)

# fit_predict: -1 = аномалия, 1 = норма
iso_predictions = iso_forest.fit_predict(X_trans)
iso_scores = iso_forest.decision_function(X_trans)  # Чем ниже, тем более аномально

# Преобразуем: -1 -> 1 (fraud), 1 -> 0 (normal)
iso_labels = (iso_predictions == -1).astype(int)

print(f"\nIsolation Forest:")
print(f"  Обнаружено аномалий: {iso_labels.sum()} ({iso_labels.mean():.2%})")
print(f"  Precision: {precision_score(y_trans, iso_labels):.4f}")
print(f"  Recall: {recall_score(y_trans, iso_labels):.4f}")
print(f"  F1: {f1_score(y_trans, iso_labels):.4f}")

# === Настройка порога по score ===
# Вместо фиксированного contamination — используем порог по score
from sklearn.metrics import roc_curve

fpr, tpr, roc_thresholds = roc_curve(y_trans, -iso_scores)  # Инвертируем: ниже score = выше вероятность fraud

# Порог для Recall >= 0.7
valid = tpr >= 0.7
if valid.any():
    best_idx = np.where(valid)[0][0]
    best_t = roc_thresholds[best_idx]
    custom_labels = (-iso_scores <= best_t).astype(int)
    print(f"\nПри пороге Recall >= 0.7:")
    print(f"  Precision: {precision_score(y_trans, custom_labels):.4f}")
    print(f"  Recall: {recall_score(y_trans, custom_labels):.4f}")
```

### Local Outlier Factor (LOF)

```python
# === LOF — учитывает локальную плотность ===
lof = LocalOutlierFactor(n_neighbors=20, contamination=0.05, novelty=False)
lof_labels = lof.fit_predict(X_trans)
lof_labels_binary = (lof_labels == -1).astype(int)

print(f"LOF:")
print(f"  Precision: {precision_score(y_trans, lof_labels_binary):.4f}")
print(f"  Recall: {recall_score(y_trans, lof_labels_binary):.4f}")
print(f"  F1: {f1_score(y_trans, lof_labels_binary):.4f}")

# === Ансамбль методов обнаружения аномалий ===
# Нормируем scores каждого метода
from sklearn.preprocessing import MinMaxScaler

scores_dict = {
    'isolation_forest': -iso_scores,
    'lof': -lof.negative_outlier_factor_,
}

scores_df = pd.DataFrame(scores_dict)
scores_df = MinMaxScaler().fit_transform(scores_df)

# Усреднённый anomaly score
ensemble_score = scores_df.mean(axis=1)

# Порог: топ-5% как аномалии
threshold = np.percentile(ensemble_score, 95)
ensemble_labels = (ensemble_score >= threshold).astype(int)

print(f"\nАнсамбль (Isolation Forest + LOF):")
print(f"  Precision: {precision_score(y_trans, ensemble_labels):.4f}")
print(f"  Recall: {recall_score(y_trans, ensemble_labels):.4f}")
print(f"  F1: {f1_score(y_trans, ensemble_labels):.4f}")
```

## 3. Снижение размерности (PCA, t-SNE, UMAP)

### PCA для кредитного скоринга

```python
from sklearn.decomposition import PCA, FactorAnalysis
from sklearn.manifold import TSNE

# === PCA: сжатие признаков для скоринга ===
np.random.seed(42)
n = 8000

# Много коррелированных признаков
credit_features = {
    'salary': np.random.lognormal(10, 0.7, n),
    'monthly_income': np.random.lognormal(10, 0.7, n) * 0.9,  # Коррелирует с salary
    'annual_income': np.random.lognormal(10, 0.7, n) * 12 * 0.95,
    'credit_limit_1': np.random.lognormal(9, 1, n),
    'credit_limit_2': np.random.lognormal(9, 1, n) * 0.8,
    'credit_limit_3': np.random.lognormal(9, 1, n) * 0.6,
    'debt_1': np.random.lognormal(7, 1.2, n),
    'debt_2': np.random.lognormal(7, 1.2, n) * 0.7,
    'spending_food': np.random.lognormal(6, 0.8, n),
    'spending_transport': np.random.lognormal(5, 0.9, n),
    'spending_entertainment': np.random.lognormal(5, 1.0, n),
    'age': np.random.randint(18, 70, n),
    'employment_years': np.random.randint(0, 30, n),
    'num_accounts': np.random.randint(1, 10, n),
}

credit_df = pd.DataFrame(credit_features)

scaler = StandardScaler()
X_credit = scaler.fit_transform(credit_df)

# === PCA ===
pca = PCA(random_state=42)
pca.fit(X_credit)

# Сколько компонент нужно для 95% дисперсии?
cumulative_variance = np.cumsum(pca.explained_variance_ratio_)
n_components_95 = np.argmax(cumulative_variance >= 0.95) + 1
print(f"Компонент для 95% дисперсии: {n_components_95} из {len(credit_features)}")
print(f"Сжатие: {len(credit_features)} -> {n_components_95} признаков")

# Топ-компоненты и их нагрузка
pca_reduced = PCA(n_components=n_components_95, random_state=42)
X_reduced = pca_reduced.fit_transform(X_credit)

print(f"\nОбъяснённая дисперсия по компонентам:")
for i, var in enumerate(pca_reduced.explained_variance_ratio_):
    print(f"  PC{i+1}: {var:.2%}")

# Интерпретация компонент через loadings
loadings = pd.DataFrame(
    pca_reduced.components_.T,
    columns=[f'PC{i+1}' for i in range(n_components_95)],
    index=credit_df.columns
)

print("\nНагрузки признаков на компоненты (топ-3 на каждую):")
for i in range(min(3, n_components_95)):
    col = f'PC{i+1}'
    top_features = loadings[col].abs().sort_values(ascending=False).head(3)
    print(f"  {col}: {', '.join(top_features.index.tolist())}")

# === Использование PCA-признаков в модели ===
from sklearn.ensemble import GradientBoostingClassifier
from sklearn.model_selection import train_test_split
from sklearn.metrics import roc_auc_score

# Целевая переменная
default_prob = (
    0.3
    - 0.0001 * credit_df['salary']
    + 0.00001 * credit_df['debt_1']
    + 0.00001 * credit_df['debt_2']
    + np.random.normal(0, 0.1, n)
)
y_credit = (default_prob > 0.3).astype(int)

X_train, X_test, y_train, y_test = train_test_split(X_credit, y_credit, test_size=0.25, random_state=42, stratify=y_credit)
X_train_pca, X_test_pca, _, _ = train_test_split(X_reduced, y_credit, test_size=0.25, random_state=42, stratify=y_credit)

# Модель на исходных признаках
model_full = GradientBoostingClassifier(n_estimators=100, random_state=42)
model_full.fit(X_train, y_train)
auc_full = roc_auc_score(y_test, model_full.predict_proba(X_test)[:, 1])

# Модель на PCA-признаках
model_pca = GradientBoostingClassifier(n_estimators=100, random_state=42)
model_pca.fit(X_train_pca, y_train)
auc_pca = roc_auc_score(y_test, model_pca.predict_proba(X_test_pca)[:, 1])

print(f"\nROC-AUC на полных признаках: {auc_full:.4f}")
print(f"ROC-AUC на PCA-признаках:    {auc_pca:.4f}")
```

## 4. Ассоциативные правила (Market Basket Analysis)

```python
# === Анализ связок финансовых продуктов ===
# Какие продукты чаще покупают вместе?

np.random.seed(42)
n_customers = 3000

products = ['checking', 'savings', 'credit_card', 'mortgage', 'investment', 'insurance', 'loan']

# Генерация покупок с корреляциями
customer_products = []
for _ in range(n_customers):
    bought = set()
    # Базовые продукты
    if np.random.random() < 0.8:
        bought.add('checking')
    if np.random.random() < 0.6:
        bought.add('savings')
    # Корреляции: mortgage -> insurance
    if 'mortgage' in bought or np.random.random() < 0.2:
        bought.add('mortgage')
        if np.random.random() < 0.7:
            bought.add('insurance')
    # Корреляции: investment -> savings
    if np.random.random() < 0.25:
        bought.add('investment')
        if np.random.random() < 0.6:
            bought.add('savings')
    # credit_card часто с checking
    if 'checking' in bought and np.random.random() < 0.5:
        bought.add('credit_card')
    # loan -> credit_card
    if np.random.random() < 0.15:
        bought.add('loan')
        if np.random.random() < 0.4:
            bought.add('credit_card')

    customer_products.append(list(bought))

# Подсчёт частот
from itertools import combinations
from collections import Counter

single_counts = Counter()
pair_counts = Counter()

for products_bought in customer_products:
    for p in products_bought:
        single_counts[p] += 1
    for pair in combinations(sorted(products_bought), 2):
        pair_counts[pair] += 1

total = len(customer_products)

print("Популярность продуктов:")
for product, count in single_counts.most_common():
    print(f"  {product}: {count/total:.1%}")

print("\nТоп-10 пар продуктов (support > 5%):")
pair_support = [(pair, count/total) for pair, count in pair_counts.most_common(20) if count/total > 0.05]
for pair, support in sorted(pair_support, key=lambda x: -x[1]):
    # Confidence: P(B|A) = support(A,B) / support(A)
    confidence_ab = support / (single_counts[pair[0]] / total)
    confidence_ba = support / (single_counts[pair[1]] / total)
    print(f"  {pair[0]} + {pair[1]}: support={support:.1%}, conf({pair[0]}->{pair[1]})={confidence_ab:.1%}, conf({pair[1]}->{pair[0]})={confidence_ba:.1%}")
```

## Практические рекомендации

1. **Выбор метода кластеризации:**
   - K-Means: быстрые, сферические кластеры, нужно знать K
   - DBSCAN: произвольная форма, автоматически находит выбросы
   - Agglomerative: иерархия, хорошо для маленьких данных

2. **Обнаружение аномалий:**
   - Isolation Forest: быстрый, работает на высоких размерностях
   - LOF: учитывает локальную плотность
   - Ансамбль методов даёт лучший результат

3. **PCA:**
   - Убирает мультиколлинеарность
   - Ускоряет обучение моделей
   - Помогает визуализировать данные
   - Не использовать t-SNE/UMAP как признаки для моделей!
