# Вебинар 5: Машинное обучение с подкреплением (Reinforcement Learning)

## Обзор темы

Обучение с подкреплением (RL) — парадигма, где агент учится принимать решения через взаимодействие со средой, получая награды. В финтехе: алгоритмическая торговля, управление портфелем, динамическое ценообразование, оптимизация кредитных решений, fraud detection в реальном времени.

## 1. Основы RL: среда, агент, награда

### Ключевые понятия
- **Agent** — алгоритм, принимающий решения
- **Environment** — среда (рынок, банк, платформа)
- **State (S)** — текущее состояние среды
- **Action (A)** — действие агента
- **Reward (R)** — награда за действие
- **Policy (π)** — стратегия агента (mapping state -> action)
- **Q-value** — ожидаемая суммарная награда за действие в состоянии

### Типы RL
- **Model-free:** Q-Learning, SARSA, DQN (не знают модель среды)
- **Model-based:** знают или строят модель среды
- **Policy Gradient:** напрямую оптимизируют политику
- **Actor-Critic:** комбинация value-based и policy-based

## 2. Q-Learning: торговый агент

```python
import numpy as np
import pandas as pd
from collections import defaultdict

# === Простая среда: торговля акцией ===
class TradingEnvironment:
    def __init__(self, prices, initial_cash=10000, commission=0.001):
        self.prices = prices
        self.initial_cash = initial_cash
        self.commission = commission
        self.reset()

    def reset(self):
        self.step_idx = 0
        self.cash = self.initial_cash
        self.shares = 0
        self.done = False
        return self._get_state()

    def _get_state(self):
        if self.step_idx >= len(self.prices) - 1:
            self.done = True
            return None
        price = self.prices[self.step_idx]
        # Состояние: [нормализованная цена, позиция, изменение цены]
        price_norm = (price - np.min(self.prices)) / (np.max(self.prices) - np.min(self.prices) + 1e-8)
        position = self.shares / 100  # Нормализация
        price_change = (self.prices[self.step_idx] / self.prices[max(0, self.step_idx - 5)] - 1) if self.step_idx >= 5 else 0
        return np.array([price_norm, position, price_change])

    def step(self, action):
        """
        Actions: 0 = Hold, 1 = Buy, 2 = Sell
        """
        if self.done:
            return None, 0, True, {}

        current_price = self.prices[self.step_idx]
        prev_portfolio_value = self.cash + self.shares * current_price

        if action == 1:  # Buy
            max_shares = int(self.cash / (current_price * (1 + self.commission)))
            shares_to_buy = min(max_shares, 10)  # Максимум 10 за раз
            cost = shares_to_buy * current_price * (1 + self.commission)
            self.cash -= cost
            self.shares += shares_to_buy

        elif action == 2:  # Sell
            shares_to_sell = min(self.shares, 10)
            revenue = shares_to_sell * current_price * (1 - self.commission)
            self.cash += revenue
            self.shares -= shares_to_sell

        self.step_idx += 1

        if self.step_idx >= len(self.prices) - 1:
            self.done = True

        new_price = self.prices[self.step_idx] if not self.done else self.prices[-1]
        portfolio_value = self.cash + self.shares * new_price
        reward = portfolio_value - prev_portfolio_value  # Изменение стоимости портфеля

        state = self._get_state()
        return state, reward, self.done, {}

    def get_final_value(self):
        return self.cash + self.shares * self.prices[-1]


# === Генерация ценового ряда ===
np.random.seed(42)
n_days = 500

# Геометрическое броуновское движение
mu = 0.0005  # Дневной дрейф
sigma = 0.02  # Волатильность
returns = np.random.normal(mu, sigma, n_days)
prices = 100 * np.cumprod(1 + returns)

print(f"Цена: {prices[0]:.2f} -> {prices[-1]:.2f} ({(prices[-1]/prices[0]-1)*100:.1f}%)")

# === Q-Learning агент ===
class QLearningAgent:
    def __init__(self, n_actions=3, learning_rate=0.1, discount=0.95, epsilon=1.0, epsilon_decay=0.995, epsilon_min=0.01):
        self.n_actions = n_actions
        self.lr = learning_rate
        self.discount = discount
        self.epsilon = epsilon
        self.epsilon_decay = epsilon_decay
        self.epsilon_min = epsilon_min
        self.q_table = defaultdict(lambda: np.zeros(n_actions))

    def discretize_state(self, state, bins=10):
        """Дискретизация непрерывного состояния"""
        discretized = []
        for i, val in enumerate(state):
            bin_idx = min(int((val + 1) / 2 * bins), bins - 1)
            bin_idx = max(0, bin_idx)
            discretized.append(bin_idx)
        return tuple(discretized)

    def choose_action(self, state):
        if np.random.random() < self.epsilon:
            return np.random.randint(self.n_actions)
        state_key = self.discretize_state(state)
        return np.argmax(self.q_table[state_key])

    def update(self, state, action, reward, next_state, done):
        state_key = self.discretize_state(state)
        if done:
            target = reward
        else:
            next_state_key = self.discretize_state(next_state)
            target = reward + self.discount * np.max(self.q_table[next_state_key])

        self.q_table[state_key][action] += self.lr * (target - self.q_table[state_key][action])

    def decay_epsilon(self):
        self.epsilon = max(self.epsilon * self.epsilon_decay, self.epsilon_min)


# === Обучение ===
n_episodes = 2000
agent = QLearningAgent()
results = []

for episode in range(n_episodes):

    state = env.reset()
    total_reward = 0

    while not env.done:

        next_state, reward, done, _ = env.step(action)

        if next_state is not None:
            agent.update(state, action, reward, next_state, done)

        total_reward += reward
        state = next_state

    agent.decay_epsilon()
    final_value = env.get_final_value()
    results.append(final_value)

    if (episode + 1) % 200 == 0:
        avg_value = np.mean(results[-200:])
        print(f"Episode {episode+1}/{n_episodes} | Avg Portfolio: ${avg_value:,.0f} | Epsilon: {agent.epsilon:.3f}")

print(f"\nОбучение завершено. Финальная стоимость портфеля: ${results[-1]:,.0f}")
print(f"Buy & Hold: ${10000 * prices[-1] / prices[0]:,.0f}")
```

## 3. Deep Q-Network (DQN) для торговли

```python
import numpy as np
import pandas as pd

# === DQN с использованием numpy (без PyTorch/TensorFlow для простоты) ===
# Полноценная реализация с нейросетью

class SimpleNeuralNetwork:
    """Простая нейросеть для Q-values (3 слоя)"""
    def __init__(self, input_size, hidden_size, output_size, learning_rate=0.001):
        self.lr = learning_rate
        # Инициализация весов (Xavier)
        self.W1 = np.random.randn(input_size, hidden_size) * np.sqrt(2.0 / input_size)
        self.b1 = np.zeros(hidden_size)
        self.W2 = np.random.randn(hidden_size, hidden_size) * np.sqrt(2.0 / hidden_size)
        self.b2 = np.zeros(hidden_size)
        self.W3 = np.random.randn(hidden_size, output_size) * np.sqrt(2.0 / hidden_size)
        self.b3 = np.zeros(output_size)

    def relu(self, x):
        return np.maximum(0, x)

    def forward(self, x):
        self.z1 = x @ self.W1 + self.b1
        self.a1 = self.relu(self.z1)
        self.z2 = self.a1 @ self.W2 + self.b2
        self.a2 = self.relu(self.z2)
        self.z3 = self.a2 @ self.W3 + self.b3
        return self.z3  # Q-values для каждого действия

    def update(self, x, target_q, action, gamma=0.95):
        """Обновление через градиентный спуск"""
        # Forward
        q_values = self.forward(x)

        # Loss: MSE между predicted Q и target Q
        target = q_values.copy()
        target[action] = target_q

        # Backward (упрощённый)
        error = target - q_values

        # Градиенты для выходного слоя
        dW3 = self.a2.reshape(-1, 1) @ error.reshape(1, -1) * self.lr
        self.W3 += dW3
        self.b3 += error * self.lr

        # Градиенты для скрытых слоёв (упрощённо)
        d_a2 = error @ self.W3.T
        d_z2 = d_a2 * (self.z2 > 0).astype(float)
        dW2 = self.a1.reshape(-1, 1) @ d_z2.reshape(1, -1) * self.lr
        self.W2 += dW2
        self.b2 += d_z2 * self.lr

        d_a1 = d_z2 @ self.W2.T
        d_z1 = d_a1 * (self.z1 > 0).astype(float)
        dW1 = x.reshape(-1, 1) @ d_z1.reshape(1, -1) * self.lr
        self.W1 += dW1
        self.b1 += d_z1 * self.lr


class DQNAgent:
    def __init__(self, state_size=10, action_size=3, hidden_size=64, lr=0.001,
                 gamma=0.95, epsilon=1.0, epsilon_decay=0.995, epsilon_min=0.01,
                 batch_size=32, memory_size=10000):
        self.action_size = action_size
        self.gamma = gamma
        self.epsilon = epsilon
        self.epsilon_decay = epsilon_decay
        self.epsilon_min = epsilon_min
        self.batch_size = batch_size
        self.memory = []
        self.memory_size = memory_size

        self.model = SimpleNeuralNetwork(state_size, hidden_size, action_size, lr)
        self.target_model = SimpleNeuralNetwork(state_size, hidden_size, action_size, lr)

    def remember(self, state, action, reward, next_state, done):
        self.memory.append((state, action, reward, next_state, done))
        if len(self.memory) > self.memory_size:
            self.memory.pop(0)

    def choose_action(self, state):
        if np.random.random() <= self.epsilon:
            return np.random.randint(self.action_size)
        q_values = self.model.forward(state)
        return np.argmax(q_values)

    def replay(self):
        if len(self.memory) < self.batch_size:
            return

        batch = [self.memory[i] for i in np.random.choice(len(self.memory), self.batch_size, replace=False)]

        for state, action, reward, next_state, done in batch:
            target = reward
            if not done:
                target = reward + self.gamma * np.max(self.target_model.forward(next_state))

            self.model.update(state, target, action)

    def update_target(self):
        self.target_model.W1 = self.model.W1.copy()
        self.target_model.b1 = self.model.b1.copy()
        self.target_model.W2 = self.model.W2.copy()
        self.target_model.b2 = self.model.b2.copy()
        self.target_model.W3 = self.model.W3.copy()
        self.target_model.b3 = self.model.b3.copy()

    def decay_epsilon(self):
        self.epsilon = max(self.epsilon * self.epsilon_decay, self.epsilon_min)


# === Расширенная среда с техническими индикаторами ===
class AdvancedTradingEnv:
    def __init__(self, prices, window_size=10, initial_cash=10000):
        self.prices = prices
        self.window_size = window_size
        self.initial_cash = initial_cash
        self.reset()

    def reset(self):
        self.step_idx = self.window_size
        self.cash = self.initial_cash
        self.shares = 0
        self.done = False
        return self._get_state()

    def _compute_features(self):
        idx = self.step_idx
        window = self.prices[max(0, idx - self.window_size):idx + 1]

        features = []
        # 1. Нормализованная цена
        features.append((self.prices[idx] - np.min(window)) / (np.max(window) - np.min(window) + 1e-8))
        # 2. SMA ratio
        sma = np.mean(window)
        features.append(self.prices[idx] / sma - 1)
        # 3. Волатильность
        features.append(np.std(window) / (np.mean(window) + 1e-8))
        # 4. RSI-подобный
        gains = np.diff(window)
        avg_gain = np.mean(gains[gains > 0]) if (gains > 0).any() else 0
        avg_loss = np.mean(-gains[gains < 0]) if (gains < 0).any() else 0.001
        rsi = avg_gain / (avg_gain + avg_loss + 1e-8)
        features.append(rsi)
        # 5. Momentum
        if len(window) > 5:
            features.append(self.prices[idx] / self.prices[idx - 5] - 1)
        else:
            features.append(0)
        # 6. Volume-like (абсолютное изменение)
        features.append(abs(np.diff(window)[-1]) / (self.prices[idx] + 1e-8))
        # 7-10. Позиция и cash
        features.append(self.shares / 100)
        features.append(self.cash / self.initial_cash)
        features.append((self.cash + self.shares * self.prices[idx]) / self.initial_cash - 1)
        # 10. Trend
        if len(window) > 3:
            features.append((window[-1] - window[0]) / (window[0] + 1e-8))
        else:
            features.append(0)

        return np.array(features)

    def _get_state(self):
        if self.step_idx >= len(self.prices) - 1:
            self.done = True
            return None
        return self._compute_features()

    def step(self, action):
        if self.done:
            return None, 0, True, {}

        current_price = self.prices[self.step_idx]
        prev_value = self.cash + self.shares * current_price

        if action == 1:  # Buy
            max_shares = int(self.cash / (current_price * 1.001))
            to_buy = min(max_shares, 20)
            self.cash -= to_buy * current_price * 1.001
            self.shares += to_buy
        elif action == 2:  # Sell
            to_sell = min(self.shares, 20)
            self.cash += to_sell * current_price * 0.999
            self.shares -= to_sell

        self.step_idx += 1
        if self.step_idx >= len(self.prices) - 1:
            self.done = True

        new_price = self.prices[self.step_idx] if not self.done else self.prices[-1]
        new_value = self.cash + self.shares * new_price
        reward = (new_value - prev_value) / prev_value  # Нормализованная награда

        return self._get_state(), reward, self.done, {}

    def get_final_value(self):
        return self.cash + self.shares * self.prices[-1]
```

## 4. RL для управления портфелем (Portfolio Optimization)

```python
# === Multi-asset Portfolio с RL ===
# Агент решает, как распределить капитал между активами

class PortfolioEnvironment:
    def __init__(self, asset_prices, rebalance_period=5, initial_capital=100000):
        """
        asset_prices: DataFrame с ценами активов (колонки = тикеры)
        rebalance_period: как часто агент может менять веса
        """
        self.asset_prices = asset_prices
        self.rebalance_period = rebalance_period
        self.initial_capital = initial_capital
        self.n_assets = asset_prices.shape[1]
        self.reset()

    def reset(self):
        self.step_idx = 0
        self.capital = self.initial_capital
        self.weights = np.ones(self.n_assets) / self.n_assets  # Равные веса
        self.done = False
        return self._get_state()

    def _get_state(self):
        if self.step_idx >= len(self.asset_prices) - self.rebalance_period:
            self.done = True
            return None

        window = self.asset_prices.iloc[max(0, self.step_idx - 20):self.step_idx + 1]

        features = []
        for i in range(self.n_assets):
            prices = window.iloc[:, i].values
            # Return за период
            ret = prices[-1] / prices[0] - 1 if len(prices) > 1 else 0
            features.append(ret)
            # Волатильность
            features.append(np.std(np.diff(prices) / prices[:-1]) if len(prices) > 1 else 0)
            # Текущий вес
            features.append(self.weights[i])

        # Sharpe-подобный признак
        all_returns = self.asset_prices.iloc[max(0, self.step_idx - 20):self.step_idx + 1].pct_change().dropna()
        if len(all_returns) > 5:
            portfolio_ret = all_returns @ self.weights
            features.append(portfolio_ret.mean() / (portfolio_ret.std() + 1e-8))
        else:
            features.append(0)

        return np.array(features)

    def step(self, action_weights):
        """
        action_weights: новый вектор весов (нормализуется автоматически)
        """
        if self.done:
            return None, 0, True, {}

        # Нормализация весов
        action_weights = np.abs(action_weights)
        weights_sum = action_weights.sum()
        if weights_sum > 0:
            new_weights = action_weights / weights_sum
        else:
            new_weights = np.ones(self.n_assets) / self.n_assets

        old_value = self.capital

        # Переход к следующему периоду
        start_idx = self.step_idx
        end_idx = min(self.step_idx + self.rebalance_period, len(self.asset_prices) - 1)

        # Доходность портфеля за период
        period_returns = self.asset_prices.iloc[end_idx] / self.asset_prices.iloc[start_idx] - 1
        portfolio_return = period_returns @ new_weights

        # Транзакционные издержки (0.1% за ребалансировку)
        turnover = np.sum(np.abs(new_weights - self.weights))
        transaction_cost = turnover * 0.001

        self.capital = self.capital * (1 + portfolio_return - transaction_cost)
        self.weights = new_weights
        self.step_idx = end_idx

        if self.step_idx >= len(self.asset_prices) - self.rebalance_period:
            self.done = True

        reward = (self.capital - old_value) / old_value
        state = self._get_state()

        return state, reward, self.done, {'portfolio_return': portfolio_return, 'turnover': turnover}

    def get_final_value(self):
        return self.capital


# === Простой RL агент для портфеля ===
class PolicyGradientAgent:
    """Упрощённый policy gradient агент"""
    def __init__(self, state_size, n_assets, learning_rate=0.01):
        self.n_assets = n_assets
        self.lr = learning_rate
        # Веса политики: state -> weights
        self.W = np.random.randn(state_size, n_assets) * 0.01
        self.b = np.zeros(n_assets)

    def get_weights(self, state):
        logits = state @ self.W + self.b
        # Softmax для получения весов
        exp_logits = np.exp(logits - np.max(logits))
        return exp_logits / exp_logits.sum()

    def update(self, state, reward):
        """Обновление в направлении градиента награды"""
        weights = self.get_weights(state)
        # Gradient: увеличиваем веса действий, которые дали положительную награду
        gradient = np.outer(state, weights) * reward
        self.W += self.lr * gradient
        self.b += self.lr * weights * reward

print(f"\nRL Portfolio avg final value: ${np.mean(portfolio_values[-50:]):,.0f}")
```

## 5. RL для кредитного одобрения

```python
# === Динамическое кредитное решение ===
# Агент решает: одобрить/отказать/запросить доп. информацию

class CreditDecisionEnv:
    def __init__(self, applicants, cost_inquiry=100, cost_default=10000, profit_loan=2000):
        self.applicants = applicants
        self.cost_inquiry = cost_inquiry
        self.cost_default = cost_default
        self.profit_loan = profit_loan
        self.reset()

    def reset(self):
        self.idx = 0
        self.total_profit = 0
        self.done = False
        return self._get_state()

    def _get_state(self):
        if self.idx >= len(self.applicants):
            self.done = True
            return None
        app = self.applicants.iloc[self.idx]
        return np.array([
            app['income'] / 100000,
            app['loan_amount'] / 500000,
            app['credit_score'] / 850,
            app['debt_to_income'],
            self.total_profit / 100000,
        ])

    def step(self, action):
        """
        0 = Отказать
        1 = Одобрить сразу
        2 = Запросить доп. информацию (стоимость inquiry)
        """
        if self.done:
            return None, 0, True, {}

        app = self.applicants.iloc[self.idx]
        will_default = app['will_default']

        if action == 0:  # Отказать
            reward = 0
        elif action == 1:  # Одобрить
            if will_default:
                reward = -self.cost_default
            else:
                reward = self.profit_loan
        elif action == 2:  # Доп. информация
            reward = -self.cost_inquiry
            # После запроса информации агент получает больше данных
            # (в упрощённой версии — просто переходим к следующему)

        self.total_profit += reward
        self.idx += 1

        if self.idx >= len(self.applicants):
            self.done = True

        return self._get_state(), reward, self.done, {}


# === Генерация данных заёмщиков ===
np.random.seed(42)
n_applicants = 1000

applicants_data = {
    'income': np.random.lognormal(10, 0.7, n_applicants),
    'loan_amount': np.random.lognormal(9, 0.8, n_applicants),
    'credit_score': np.random.randint(300, 850, n_applicants),
    'debt_to_income': np.random.uniform(0, 0.7, n_applicants),
}

app_df = pd.DataFrame(applicants_data)

# Вероятность дефолта
default_prob = (
    0.4
    - 0.0001 * app_df['income']
    + 0.000005 * app_df['loan_amount']
    - 0.001 * app_df['credit_score']
    + 0.5 * app_df['debt_to_income']
    + np.random.normal(0, 0.1, n_applicants)
)
app_df['will_default'] = (default_prob > 0.3).astype(int)

print(f"Default rate: {app_df['will_default'].mean():.2%}")

# === Обучение агента ===
credit_agent = QLearningAgent(n_actions=3, epsilon=0.3, epsilon_decay=0.999)

n_episodes = 500
for episode in range(n_episodes):
    env = CreditDecisionEnv(app_df)
    state = env.reset()

    while not env.done:
        action = credit_agent.choose_action(state)
        next_state, reward, done, _ = env.step(action)

        if next_state is not None:
            credit_agent.update(state, action, reward, next_state, done)

        state = next_state

    credit_agent.decay_epsilon()

    if (episode + 1) % 100 == 0:
        print(f"Episode {episode+1}/{n_episodes} | Total Profit: ${env.total_profit:,.0f}")

print(f"\nRL агент завершил с прибылью: ${env.total_profit:,.0f}")

# === Сравнение с простым правилом ===
# Правило: одобрить если credit_score > 650 и debt_to_income < 0.4
simple_profit = 0
for _, app in app_df.iterrows():
    if app['credit_score'] > 650 and app['debt_to_income'] < 0.4:
        if app['will_default']:
            simple_profit -= 10000
        else:
            simple_profit += 2000

print(f"Простое правило прибыль: ${simple_profit:,.0f}")
```

## Практические рекомендации

1. **Когда применять RL в финтехе:**
   - Последовательные решения (торговля, кредитование)
   - Динамическая среда (меняющиеся рынки)
   - Долгосрочная оптимизация (портфель)

2. **Проблемы и решения:**
   - Нестационарность:定期 переобучение, online learning
   - Risk management: ограничение позиций, stop-loss
   - Simulation gap: тестирование на исторических данных (backtesting)

3. **Практические ограничения:**
   - RL требует много данных и вычислений
   - Сложно интерпретировать решения
   - В продакшене — гибридные подходы (RL + правила)
